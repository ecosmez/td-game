(fn ShowBuildHUD ()
  (bind pc (Game|GetPlayerController 0))
  (Utilities|IsValid pc
    (:"Is Valid"
      ; C++ helper — no Create Widget Class pin to go missing at compile time.
      (bind widget (TD|UI|CreateAndShowTowerStore :OwningPlayer pc :ZOrder 120))
      (Utilities|IsValid widget
        (:"Is Valid"
          (Variables|Default|SetBuildHUDWidget widget))
        (:"Is Not Valid"
          (Development|PrintString "ShowBuildHUD: CreateAndShowTowerStore failed"))))))
