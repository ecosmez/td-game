(fn BeginTowerDrag ()
  (if (Variables|Default|GetIsBuildPhase)
    (if (>= (Variables|Resource|GetResource) (Variables|Store|GetSelectedTowerCost))
      (Variables|Default|SetIsDragging true)
      (Variables|Default|SetTowerDragLive true)
      (Variables|Default|SetHasTowerSelected true)
      (Variables|Default|SetStatusLine "Dragging tower - aim at a hex then LMB or RMB")
      (Variables|Default|SetHintLine "LMB/RMB on pad places | ESC cancels")
      (CallFunction|ShowHexGrid)
      (CallFunction|SpawnGhostTower))))
