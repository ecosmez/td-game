(event EventTick (DeltaSeconds)
  (bind _returnvalue (Game|GetPlayerController 0))
  (bind _springarm (Variables|Camera|GetSpringArm))
  (bind _targetarmlength (Class|SpringArmComponent|GetTargetArmLength _springarm))
  (bind _zoomspeed (Variables|Default|GetZoomSpeed))
  (if (|GetbIsDropping)
    (Utilities|IsValid _returnvalue
      (:"Is Valid"
        (if (Game|Player|GetHitResultUnderCursorByChannel _returnvalue)
          (Pawn|Input|AddMovementInput self (Math|Vector|GetUnitDirection(Vector) (Math|Vector|MakeVector (.x (Transformation|GetActorLocation)) (.y (Transformation|GetActorLocation))) (Math|Vector|MakeVector (.x (Collision|BreakHitResult (Game|Player|GetHitResultUnderCursorByChannel _returnvalue))) (.y (Collision|BreakHitResult (Game|Player|GetHitResultUnderCursorByChannel _returnvalue))))) (Variables|Default|GetSteerScale)))
        (if (Game|Player|IsInputKeyDown _returnvalue "SpaceBar")
          (Class|ABPUnarmed|SetVelocity (Math|Vector|MakeVector (.x (Transformation|GetVelocity)) (.y (Transformation|GetVelocity)) (Variables|Default|GetGlideFallSpeed)) (Variables|Character|GetCharacterMovement)))))
    (else
      (Variables|Default|SetAbilityDelta DeltaSeconds)
      (CallFunction|TickAbilities)
      (Utilities|IsValid _returnvalue
        (:"Is Valid"
          (if (Game|Player|WasInputKeyJustPressed _returnvalue "MouseScrollUp")
            (Class|SpringArmComponent|SetTargetArmLength (select (< (- _targetarmlength _zoomspeed) (Variables|Default|GetMinZoom)) (Variables|Default|GetMinZoom) (- _targetarmlength _zoomspeed)) _springarm))
          (if (Game|Player|WasInputKeyJustPressed _returnvalue "MouseScrollDown")
            (Class|SpringArmComponent|SetTargetArmLength (select (> (+ _targetarmlength _zoomspeed) (Variables|Default|GetMaxZoom)) (Variables|Default|GetMaxZoom) (+ _targetarmlength _zoomspeed)) _springarm))
          (if (<= (Variables|Default|GetPendingAbility) 0)
            (if (and (Game|Player|WasInputKeyJustPressed _returnvalue "Q") (<= (|GetCD_Q) 0.0))
              (CallFunction|BeginAbilityAim 1))
            (if (and (Game|Player|WasInputKeyJustPressed _returnvalue "W") (<= (|GetCD_W) 0.0))
              (CallFunction|BeginAbilityAim 2))
            (if (and (Game|Player|WasInputKeyJustPressed _returnvalue "E") (<= (|GetCD_E) 0.0))
              (CallFunction|BeginAbilityAim 3))
            (if (and (Game|Player|WasInputKeyJustPressed _returnvalue "R") (<= (|GetCD_R) 0.0))
              (CallFunction|BeginAbilityAim 4)))
          (if (> (Variables|Default|GetPendingAbility) 0)
            (CallFunction|UpdateAbilityAim)
            (if (Game|Player|WasInputKeyJustPressed _returnvalue "LeftMouseButton")
              (CallFunction|ConfirmAbilityAim))
            (if (Game|Player|WasInputKeyJustPressed _returnvalue "Escape")
              (CallFunction|CancelAbilityAim))))))))

(event EventBeginPlay
  (bind _returnvalue (Game|GetPlayerController 0))
  (bind _charactermovement (Variables|Character|GetCharacterMovement))
  (bind _springarm (Variables|Camera|GetSpringArm))
  (|SetbIsDropping true)
  (Transformation|SetActorLocation self (Math|Vector|MakeVector (.x (Transformation|GetActorLocation)) (.y (Transformation|GetActorLocation)) (Variables|Default|GetDropHeight)) false true)
  (Class|CharacterMovementComponent|SetGravityScale (Variables|Default|GetFreefallGravityScale) _charactermovement)
  (Class|CharacterMovementComponent|SetAirControl (Variables|Default|GetAirControlStrength) _charactermovement)
  (Variables|Default|SetSavedArmLength (Class|SpringArmComponent|GetTargetArmLength _springarm))
  (Class|SpringArmComponent|SetTargetArmLength (Variables|Default|GetSavedArmLength) _springarm)
  (Transformation|SetRelativeRotation _springarm (Math|Rotator|MakeRotator 0.0 -90.0))
  (Utilities|IsValid _returnvalue
    (:"Is Valid"
      (|SetbShowMouseCursor true _returnvalue)
      (|SetbEnableClickEvents true _returnvalue)
      (|SetbEnableMouseOverEvents true _returnvalue)
      (bind _asmoba_player_controller (Utilities|Casting|CastToMobaPlayerController (Pawn|GetController)))
      (|SetbEnableClickToMoveChampion false _asmoba_player_controller))))

(event EventOnLanded (Hit)
  (bind _charactermovement (Variables|Character|GetCharacterMovement))
  (bind _springarm (Variables|Camera|GetSpringArm))
  (if (|GetbIsDropping)
    (|SetbIsDropping false)
    (Class|CharacterMovementComponent|SetGravityScale (Variables|Default|GetDefaultGravityScale) _charactermovement)
    (Class|CharacterMovementComponent|SetAirControl (Variables|Default|GetDefaultAirControl) _charactermovement)
    (Class|SpringArmComponent|SetTargetArmLength (Variables|Default|GetSavedArmLength) _springarm)
    (Transformation|SetRelativeRotation _springarm (Math|Rotator|MakeRotator 0.0 (Variables|Default|GetTopDownArmPitch)))
    (bind _asmoba_player_controller (Utilities|Casting|CastToMobaPlayerController (Pawn|GetController))
      (:then
        (|SetbEnableClickToMoveChampion true _asmoba_player_controller)
        (CallFunction|ShowAbilityHUD))
      (:CastFailed
        (CallFunction|ShowAbilityHUD)))))
