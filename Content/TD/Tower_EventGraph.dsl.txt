(event EventBeginPlay
  (Variables|Default|SetCurrentHealth (Variables|Default|GetMaxHealth))
  (if (> (Variables|Combat|GetFireCooldown) 0.0)
    (Variables|Combat|SetFireInterval (Variables|Combat|GetFireCooldown))
    (elif (> (Variables|Combat|GetAttackSpeed) 0.0)
      (Variables|Combat|SetFireInterval (/ 1.0 (Variables|Combat|GetAttackSpeed)))))
  (CallFunction|GatherFirePoints)
  (Variables|Construction|SetMeshMinZ (.z (Components|StaticMesh|GetLocalBounds (Variables|Default|GetTowerMesh))))
  (Rendering|SetVisibility (Variables|Default|GetHoloMesh))
  (Variables|Construction|SetIsConstructing false)
  (Variables|Construction|SetIsBuilt false)
  (Variables|Construction|SetConstructionProgress 0.0)
  (CallFunction|UpdateTowerHealthBar))

(event Collision|EventActorBeginOverlap (OtherActor))

(event EventTick (DeltaSeconds)
  (CallFunction|UpdateTowerHealthBar)
  (if (Variables|Construction|GetIsBuilt)
    (if (Variables|Default|GetIsAoe)
      (CallFunction|UpdateAoEBehavior self DeltaSeconds)
      (else
        (CallFunction|TryFire)))
    (elif (Variables|Construction|GetIsConstructing)
      (CallFunction|UpdateConstruction))))

; FireCooldown > 0 copies into FireInterval; else FireInterval = 1/AttackSpeed.
; IsAoe then → UpdateAoEBehavior (PulseInterval). else → TryFire (FireCooldown / FireInterval).
; See Tower_UpdateAoEBehavior.dsl.txt.
