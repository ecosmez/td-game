; Per-fire-point combat. See also GatherFirePoints / SelectVisibleTarget / SpawnVolleyAt.
(fn TryFire ()
  (if (not (Variables|Default|GetCanAttack))
    (return))
  (if (<= (Variables|Combat|GetAttackSpeed) 0.0)
    (return))
  (bind dt (Utilities|Time|GetWorldDeltaSeconds))
  (bind n (Utilities|Array|Length (Variables|Combat|GetFirePoints)))
  (for i (range n)
    (bind cd (- (Utilities|Array|Get(aref) (Variables|Combat|GetFirePointCooldowns) i) dt))
    (Utilities|Array|SetArrayElem :TargetArray (Variables|Combat|GetFirePointCooldowns) :Index i :Item cd)
    (if (<= cd 0.0)
      (bind fp (Utilities|Array|Get(aref) (Variables|Combat|GetFirePoints) i))
      (Utilities|IsValid fp
        (:"Is Valid"
          (bind origin (Transformation|GetWorldLocation fp))
          (CallFunction|SelectVisibleTarget :Origin origin)
          (if (Variables|Combat|GetHasFireTarget)
            (Utilities|Array|SetArrayElem :TargetArray (Variables|Combat|GetFirePointCooldowns) :Index i :Item (Variables|Combat|GetFireInterval))
            (CallFunction|SpawnVolleyAt :Origin origin)))
        (:"Is Not Valid")))))

; SelectVisibleTarget(Origin): nearest in-Range enemy with clear Visibility LoS.
; Trace ActorsToIgnore = all enemies; bIgnoreSelf false (tower body blocks).
; Sets BestFireTarget / BestFireDist / HasFireTarget.

; SpawnVolleyAt(Origin): for s in 0..max(ShotsPerVolley,1)-1 spawn BP_Projectile at
; (Origin.X + s*40, Origin.Y, Origin.Z), InitProjectile(BestFireTarget, DamagePerShot).
