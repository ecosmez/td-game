(fn AdvanceAlongPath (DeltaSeconds)
  (bind _self self)
  (bind _returnvalue (Transformation|GetActorLocation _self))
  (bind _crystalactor (Variables|Default|GetCrystalActor))
  (bind _returnvalue_1 (Transformation|GetActorLocation _crystalactor))
  (bind _waypointindex (Variables|Default|GetWaypointIndex))
  (bind _waypoints (Variables|Default|GetWaypoints))
  (bind _output (Utilities|Array|Get(acopy) _waypoints _waypointindex))
  (bind _x (.x _output))
  (bind _y (.y _output))
  (bind _z (.z (CallFunction|SnaptoGround (Math|Interpolation|VInterpTo_Constant _returnvalue _output (Variables|Default|GetLastDelta) (* (Variables|Default|GetMoveSpeed) (Variables|Default|GetSlowFactor))))))
  (bind _returnvalue_2 (Math|Vector|MakeVector _x _y _z))
  (bind _x_1 (.x _returnvalue_1))
  (bind _y_1 (.y _returnvalue_1))
  (bind _z_1 (.z (CallFunction|SnaptoGround (Math|Interpolation|VInterpTo_Constant _returnvalue _returnvalue_1 (Variables|Default|GetLastDelta) (* (Variables|Default|GetMoveSpeed) (Variables|Default|GetSlowFactor))))))
  (bind _returnvalue_3 (Math|Vector|MakeVector _x_1 _y_1 _z_1))
  (bind _slowfactor (Variables|Default|GetSlowFactor))
  (bind _movespeed (Variables|Default|GetMoveSpeed))
  (bind _returnvalue_4 (* _movespeed _slowfactor))
  (bind _attackcooldown (Variables|Default|GetAttackCooldown))
  (bind _returnvalue_5 (- _attackcooldown DeltaSeconds))
  (bind _lastdelta (Variables|Default|GetLastDelta))
  (if (Variables|Default|GetIsAttackingCrystal)
    (if (<= (Variables|Default|GetStunRemaining) 0.0)
      (Variables|Default|SetAttackCooldown _returnvalue_5)
      (if (<= _returnvalue_5 0.0)
        (CallFunction|AttackCrystal))))
  (if (not (Variables|Default|GetIsAttackingCrystal))
    (if (Variables|Default|GetIsWallBlocked)
      (if (<= (Variables|Default|GetStunRemaining) 0.0)
        (CallFunction|AttackBlockingWall DeltaSeconds))
      (elif (<= (Variables|Default|GetStunRemaining) 0.0)
        (if (Utilities|Array|IsEmpty (Variables|Default|GetWaypoints))
          (CallFunction|ChoosePath))
        (if (Utilities|Array|IsValidIndex _waypoints _waypointindex)
          (bind _snappedlocation (CallFunction|SnaptoGround (Math|Interpolation|VInterpTo_Constant _returnvalue _output _lastdelta _returnvalue_4)))
          (Transformation|SetActorLocation _self _snappedlocation false true)
          (Transformation|SetActorRotation _self (Math|Rotator|FindLookAtRotation _returnvalue _returnvalue_2) true)
          (if (<= (Math|Vector|Distance(Vector) _snappedlocation _returnvalue_2) 120.0)
            (CallFunction|AdvanceWaypoint))
          (else
            (Utilities|IsValid _crystalactor
              (:"Is Valid"
                (bind _snappedlocation (CallFunction|SnaptoGround (Math|Interpolation|VInterpTo_Constant _returnvalue _returnvalue_1 _lastdelta _returnvalue_4)))
                (Transformation|SetActorLocation _self _snappedlocation false true)
                (Transformation|SetActorRotation _self (Math|Rotator|FindLookAtRotation _returnvalue _returnvalue_3) true)
                (if (<= (Math|Vector|Distance(Vector) _snappedlocation _returnvalue_3) 200.0)
                  (CallFunction|ReachCrystal)))
              (:"Is Not Valid"
                (CallFunction|ReachCrystal)))))))))
