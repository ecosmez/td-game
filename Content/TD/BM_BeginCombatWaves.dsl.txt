(fn BeginCombatWaves ()
  (if (not (Variables|Default|GetWavesStarted))
    (Variables|Default|SetWavesStarted true)
    (CallFunction|DestroyGhostTower)
    (CallFunction|HideHexGrid)
    (Variables|Default|SetHasTowerSelected false)
    (Variables|Default|SetIsDragging false)
    (Variables|Default|SetTowerDragLive false)
    (Variables|Default|SetStatusLine "WAVES STARTED - keep building!")
    (Variables|Default|SetHintLine "Drag towers anytime | ENTER force starts next wave | ESC cancels")
    (bind _outactors (Actor|GetAllActorsOfClass "/Game/TD/BP_EnemySpawner.BP_EnemySpawner_C"))
    (if (not (Utilities|Array|IsEmpty _outactors))
      (for _array_element _outactors
        (bind _asbp_enemy_spawner (Utilities|Casting|CastToBP_EnemySpawner _array_element))
        (CallFunction|StartWaves _asbp_enemy_spawner)))))
