(fn TryDropTower ()
  (if (Variables|Default|GetIsBuildPhase)
    (if (Variables|Default|GetHasTowerSelected)
      (if (or
            (Utilities|String|EqualExactly(String) (Variables|Store|GetSelectedTowerClassPath) "/Game/TD/Traps/BP_Trap_Base.BP_Trap_Base_C")
            (Utilities|String|EqualExactly(String) (Variables|Store|GetSelectedTowerClassPath) "/Game/TD/Towers/BP_Tower_Mine.BP_Tower_Mine_C"))
        (bind pc (Game|GetPlayerController 0))
        (Utilities|IsValid pc
          (:"Is Valid"
            (bind (hitresult didhit) (Game|Player|GetHitResultUnderCursorByChannel pc "Visibility" true))
            (if didhit
              (bind (blockinghit initialoverlap hittime hitdistance hitlocation impactpoint) (Collision|BreakHitResult hitresult))
              (bind pathdistance (TD|EnemyPath|GetDistanceToNearestPath impactpoint))
              (if (<= pathdistance (Variables|Store|GetSelectedTowerRadius))
                (if (>= (Variables|Resource|GetResource) (Variables|Store|GetSelectedTowerCost))
                  (CallFunction|TrySpendResource)
                  (bind ghost (Variables|Default|GetGhostTower))
                  (bind spawnlocation (CallFunction|GroundSnapFromGhost :Surface impactpoint :Ghost ghost))
                  (bind spawnedactor (CallFunction|SpawnandStartTower :self self :SpawnXF (Math|Transform|MakeTransform spawnlocation)))
                  (Utilities|IsValid spawnedactor
                    (:"Is Valid"
                      (CallFunction|FinishTowerPlacement))))))))
        (else
          (if (Utilities|String|EqualExactly(String) (Variables|Store|GetSelectedTowerClassPath) "/Game/TD/Towers/BP_Tower_Wall.BP_Tower_Wall_C")
            (bind pc (Game|GetPlayerController 0))
            (Utilities|IsValid pc
              (:"Is Valid"
                (bind (hitresult didhit) (Game|Player|GetHitResultUnderCursorByChannel pc "Visibility" true))
                (if didhit
                  (bind (blockinghit initialoverlap hittime hitdistance hitlocation impactpoint Normal ImpactNormal PhysMat HitActor) (Collision|BreakHitResult hitresult))
                  (if (Variables|Default|GetWallRotationPending)
                    (if (>= (Variables|Resource|GetResource) (Variables|Store|GetSelectedTowerCost))
                      (CallFunction|TrySpendResource)
                      (bind locked (Variables|Default|GetWallLockedLocation))
                      (bind look (Math|Rotator|FindLookAtRotation locked impactpoint))
                      (bind yawrot (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw (.yaw look)))
                      (bind spawnedactor (CallFunction|SpawnandStartTower :self self :SpawnXF (Math|Transform|MakeTransform locked yawrot)))
                      (Utilities|IsValid spawnedactor
                        (:"Is Valid"
                          (CallFunction|FinishTowerPlacement))))
                    (else
                      (if (TD|Abilities|IsValidAbilityTerrainActor HitActor)
                        (if (>= (Variables|Resource|GetResource) (Variables|Store|GetSelectedTowerCost))
                          (bind ghost (Variables|Default|GetGhostTower))
                          (bind spawnlocation (CallFunction|GroundSnapFromGhost :Surface impactpoint :Ghost ghost))
                          (Variables|Default|SetWallLockedLocation spawnlocation)
                          (Variables|Default|SetWallRotationPending true)
                          (Variables|Default|SetStatusLine "Wall position locked - rotate facing")
                          (Variables|Default|SetHintLine "Yellow arrow is facing | LMB places | ESC / RMB cancels")
                          (Utilities|IsValid ghost
                            (:"Is Valid"
                              (Transformation|SetActorLocation ghost spawnlocation false true))))))))))
            (else
              (CallFunction|ResolveCursorPad))))))))
