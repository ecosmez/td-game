(fn SpawnEnemyInner ()
  (bind trashHp (+ (Variables|Default|GetBaseEnemyHP) (* (Variables|Default|GetHPPerWave) (- (Variables|Default|GetWaveNumber) 1))))
  (bind pathactor (Variables|Default|GetPathActor))
  (bind spawnBoss (and (Variables|Default|GetIsBossWave) (>= (Variables|Default|GetWaveSpawnedCount) (Variables|Default|GetNormalEnemiesThisWave))))
  (bind spawnRanged (and (not spawnBoss) (and (>= (Variables|Default|GetWaveNumber) (Variables|Default|GetRangedStartWave)) (== (Math|Integer|%(Integer) (Variables|Default|GetWaveSpawnedCount) 3) 0))))
  (Utilities|IsValid pathactor
    (:"Is Valid"
      (bind aspath (Utilities|Casting|CastToBP_EnemyPath pathactor)
        (:then
          (bind spawnXform (Math|Transform|MakeTransform (select (Utilities|Array|IsValidIndex (Class|BPEnemyPath|GetWaypoints aspath)) (Utilities|Array|Get(acopy) (Class|BPEnemyPath|GetWaypoints aspath)) (Transformation|GetActorLocation aspath))))
          (if spawnBoss
            (bind spawned (Game|SpawnActorfromClass "/Game/TD/BP_Boss.BP_Boss_C" spawnXform "AdjustIfPossibleButAlwaysSpawn" "MultiplyWithRoot"))
            (Utilities|IsValid spawned
              (:"Is Valid"
                (bind enemy (Utilities|Casting|CastToBP_Enemy spawned)
                  (:then
                    (Class|BPEnemy|SetPathActor :PathActor pathactor :self enemy)
                    (Class|BPEnemy|SetCrystalActor :CrystalActor (Variables|Default|GetCrystalActor) :self enemy)
                    (Class|BPEnemy|SetRouteId :routeId (Variables|Default|GetRouteId) :self enemy)
                    (Class|BPEnemy|SetUseLanePreference true enemy)
                    (Class|BPEnemy|SetPreferOverLane (select (>= (Variables|Default|GetWaveNumber) (Variables|Default|GetAlternatePathStartWave)) (== (Math|Integer|%(Integer) (Variables|Default|GetWaveSpawnedCount) 2) 0) true) enemy)
                    (Class|BPEnemy|BuildNavPath enemy)
                    (Variables|Default|SetWaveSpawnedCount (+ (Variables|Default|GetWaveSpawnedCount) 1))
                    (Development|PrintString "Siege Brute spawned!" true true "(R=1.000000,G=0.250000,B=0.100000,A=1.000000)" 3.0))
                  (:CastFailed
                    (Development|PrintString "Boss spawn cast failed"))))
              (:"Is Not Valid"
                (Development|PrintString "Boss SpawnActor null")))
            (else
              (if spawnRanged
                (bind spawnedR (Game|SpawnActorfromClass "/Game/TD/BP_RangedEnemy.BP_RangedEnemy_C" spawnXform "AdjustIfPossibleButAlwaysSpawn" "MultiplyWithRoot"))
                (Utilities|IsValid spawnedR
                  (:"Is Valid"
                    (bind enemy (Utilities|Casting|CastToBP_Enemy spawnedR)
                      (:then
                        (Class|BPEnemy|SetPathActor :PathActor pathactor :self enemy)
                        (Class|BPEnemy|SetCrystalActor :CrystalActor (Variables|Default|GetCrystalActor) :self enemy)
                        (Class|BPEnemy|SetRouteId :routeId (Variables|Default|GetRouteId) :self enemy)
                        (Class|BPEnemy|SetMaxHealth trashHp enemy)
                        (Class|BPEnemy|SetCurrentHealth trashHp enemy)
                        (Class|BPEnemy|SetUseLanePreference true enemy)
                        (Class|BPEnemy|SetPreferOverLane (select (>= (Variables|Default|GetWaveNumber) (Variables|Default|GetAlternatePathStartWave)) (== (Math|Integer|%(Integer) (Variables|Default|GetWaveSpawnedCount) 2) 0) true) enemy)
                        (Class|BPEnemy|BuildNavPath enemy)
                        (Variables|Default|SetWaveSpawnedCount (+ (Variables|Default|GetWaveSpawnedCount) 1))
                        (Development|PrintString "Ranged enemy spawned!" true true "(R=0.200000,G=0.850000,B=0.300000,A=1.000000)" 2.0))
                      (:CastFailed
                        (Development|PrintString "Ranged spawn cast failed"))))
                  (:"Is Not Valid"
                    (Development|PrintString "Ranged SpawnActor null")))
                (else
                  (bind spawned (Game|SpawnActorfromClass "/Game/TD/BP_Enemy.BP_Enemy_C" spawnXform "AdjustIfPossibleButAlwaysSpawn" "MultiplyWithRoot"))
                  (Utilities|IsValid spawned
                    (:"Is Valid"
                      (bind enemy (Utilities|Casting|CastToBP_Enemy spawned)
                        (:then
                          (Class|BPEnemy|SetPathActor :PathActor pathactor :self enemy)
                          (Class|BPEnemy|SetCrystalActor :CrystalActor (Variables|Default|GetCrystalActor) :self enemy)
                          (Class|BPEnemy|SetRouteId :routeId (Variables|Default|GetRouteId) :self enemy)
                          (Class|BPEnemy|SetMaxHealth trashHp enemy)
                          (Class|BPEnemy|SetCurrentHealth trashHp enemy)
                          (Class|BPEnemy|SetUseLanePreference true enemy)
                          (Class|BPEnemy|SetPreferOverLane (select (>= (Variables|Default|GetWaveNumber) (Variables|Default|GetAlternatePathStartWave)) (== (Math|Integer|%(Integer) (Variables|Default|GetWaveSpawnedCount) 2) 0) true) enemy)
                          (Class|BPEnemy|BuildNavPath enemy)
                          (Variables|Default|SetWaveSpawnedCount (+ (Variables|Default|GetWaveSpawnedCount) 1)))
                        (:CastFailed
                          (Development|PrintString "Spawn cast failed"))))
                    (:"Is Not Valid"
                      (Development|PrintString "SpawnActor null"))))))))
        (:CastFailed
          (Development|PrintString "PathActor is not BP_EnemyPath"))))
    (:"Is Not Valid"
      (Development|PrintString "Spawner missing PathActor"))))
