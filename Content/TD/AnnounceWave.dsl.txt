(fn AnnounceWave ()
  (Variables|Default|SetWaveNumber (+ (Variables|Default|GetWaveNumber) 1))
  (Variables|Default|SetWaveSpawnedCount 0)
  (Variables|Default|SetIsSpawningWave false)
  (Variables|Default|SetWaitingforClear false)
  (Variables|Default|SetNormalEnemiesThisWave (+ 4 (* (Variables|Default|GetWaveNumber) 2)))
  (Variables|Default|SetIsBossWave (and (> (Variables|Boss|GetBossWaveNumber) 0) (== (Math|Integer|%(Integer) (Variables|Default|GetWaveNumber) (Variables|Boss|GetBossWaveNumber)) 0)))
  (Variables|Default|SetEnemiesPerWave (+ (Variables|Default|GetNormalEnemiesThisWave) (select (Variables|Default|GetIsBossWave) 1 0)))
  (if (Variables|Default|GetIsBossWave)
    (Development|PrintString (Utilities|String|BuildString(Integer) "" "=== BOSS WAVE " (Variables|Default|GetWaveNumber) " - SIEGE BRUTE ===") true true "(R=1.000000,G=0.200000,B=0.100000,A=1.000000)" 5.0)
    (else
      (Development|PrintString (Utilities|String|BuildString(Integer) "" "=== WAVE " (Variables|Default|GetWaveNumber) " INCOMING ===") true true "(R=1.000000,G=0.850000,B=0.100000,A=1.000000)" 4.0)))
  (Development|PrintString (Utilities|String|Append "Enemies: " (Utilities|String|ToString(Integer) (Variables|Default|GetEnemiesPerWave))) true true "(R=1.000000,G=0.850000,B=0.100000,A=1.000000)" 4.0)
  (Development|PrintString (Utilities|String|Append "Enemy HP: " (Utilities|String|ToString(Float) (+ (Variables|Default|GetBaseEnemyHP) (* (Variables|Default|GetHPPerWave) (- (Variables|Default|GetWaveNumber) 1))))) true true "(R=1.000000,G=0.850000,B=0.100000,A=1.000000)" 4.0)
  (Variables|Default|SetCountdownRemaining 5)
  (Utilities|Time|SetTimerbyFunctionName self "TickCountdown" 1.0 true)
  (CallFunction|TickCountdown))
