(fn TickCountdown ()
  (bind _countdownremaining (Variables|Default|GetCountdownRemaining))
  (if (<= _countdownremaining 0)
    (Utilities|Time|ClearTimerbyFunctionName self "TickCountdown")
    (CallFunction|StartWaveSpawning)
    (else
      (CallFunction|ShowCountdown)
      (Variables|Default|SetCountdownRemaining (- _countdownremaining 1)))))
