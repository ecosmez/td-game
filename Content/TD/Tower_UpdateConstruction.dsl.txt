(fn UpdateConstruction ()
  (bind _returnvalue (Utilities|Time|GetWorldDeltaSeconds))
  (bind _constructiontime (Variables|Construction|GetConstructionTime))
  (bind _returnvalue_1 (/ _returnvalue _constructiontime))
  (bind _constructionprogress (Variables|Construction|GetConstructionProgress))
  (bind _returnvalue_2 (+ _constructionprogress _returnvalue_1))
  (if (>= _returnvalue_2 1.0)
    (Variables|Construction|SetConstructionProgress 1.0)
    (Variables|Construction|SetIsConstructing false)
    (Variables|Construction|SetIsBuilt true)
    (Rendering|SetVisibility (Variables|Default|GetHoloMesh))
    (else
      (Variables|Construction|SetConstructionProgress _returnvalue_2))))
