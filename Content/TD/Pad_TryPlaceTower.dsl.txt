(fn TryPlaceTower ()
  (bind loc (Transformation|GetActorLocation))
  (bind spawn_xf (Math|Transform|MakeTransform (Math|Vector|MakeVector (.x loc) (.y loc) (+ (.z loc) 120.0))))
  (bind existing (Variables|Default|GetPlacedTower))
  (bind managers (Actor|GetAllActorsOfClass "/Game/TD/BP_BuildManager.BP_BuildManager_C"))
  (if (Utilities|Array|IsEmpty managers)
    (Development|PrintString "No BuildManager" true true "(R=1.000000,G=0.200000,B=0.200000,A=1.000000)" 3.0)
    (else
      (bind mgr (Utilities|Casting|CastToBP_BuildManager (Utilities|Array|RandomArrayItem managers)))
      (if (Class|BPBuildManager|GetIsBuildPhase mgr)
        (if (Class|BPBuildManager|GetHasTowerSelected mgr)
          (Utilities|FlowControl|Sequence
            (:then_0
              (Utilities|IsValid existing
                (:"Is Valid"
                  (Actor|DestroyActor existing))))
            (:then_1
              (bind spawned (Game|SpawnActorfromClass "/Game/TD/BP_Tower.BP_Tower_C" spawn_xf "AlwaysSpawn" "MultiplyWithRoot")
                (:then
                  (Variables|Default|SetPlacedTower spawned)
                  (bind as_tower (Utilities|Casting|CastToBP_Tower spawned)
                    (:then
                      (Class|BPTower|StartConstruction as_tower))
                    (:CastFailed))))))
          (else
            (Development|PrintString "Select a tower first" true true "(R=1.000000,G=0.700000,B=0.200000,A=1.000000)" 3.0)))
        (else
          (Development|PrintString "Waves already started" true true "(R=1.000000,G=0.500000,B=0.200000,A=1.000000)"))))))
