(fn BeginAbilityAim (AbilityId)
  (if (and (> (Variables|Default|GetPendingAbility) 0) (== (Variables|Default|GetPendingAbility) AbilityId))
    (CallFunction|CancelAbilityAim)
    (else
      (CallFunction|CancelAbilityAim)
      (Variables|Default|SetPendingAbility AbilityId)
      (bind _returnvalue (Game|SpawnActorBPAbilityAimPreview "/Game/TD/BP_AbilityAimPreview.BP_AbilityAimPreview_C" (Math|Transform|MakeTransform (Transformation|GetActorLocation)) "AlwaysSpawn" "MultiplyWithRoot" 0 0))
      (Variables|Default|SetAimPreviewActor _returnvalue)
      (Collision|SetActorEnableCollision _returnvalue)
      (CallFunction|UpdateAbilityAim))))
