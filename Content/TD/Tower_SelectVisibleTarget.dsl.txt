(fn SelectVisibleTarget (Origin)
  (Variables|Combat|SetHasFireTarget false)
  (Variables|Combat|SetBestFireDist 999999.0)
  (bind enemies (Actor|GetAllActorsOfClass "/Game/TD/BP_Enemy.BP_Enemy_C"))
  (bind range (Variables|Combat|GetRange))
  (for enemy enemies
    (Utilities|IsValid enemy
      (:"Is Valid"
        (bind enemy_loc (Transformation|GetActorLocation enemy))
        (bind dist (Math|Vector|Distance(Vector) Origin enemy_loc))
        (if (and (<= dist range) (< dist (Variables|Combat|GetBestFireDist)))
          (bind dir (Math|Vector|GetUnitDirection(Vector) Origin enemy_loc))
          (bind start (+ Origin (* dir 10.0)))
          (bind end (+ enemy_loc (Math|Vector|MakeVector 0.0 0.0 50.0)))
          (bind (outhit blocked) (Collision|LineTraceByChannel
            :Start start
            :End end
            :TraceChannel "Visibility"
            :ActorsToIgnore enemies
            :bIgnoreSelf false
            :DrawDebugType "None"))
          (if (not blocked)
            (Variables|Combat|SetBestFireDist dist)
            (Variables|Combat|SetBestFireTarget enemy)
            (Variables|Combat|SetHasFireTarget true))))
      (:"Is Not Valid"))))
