(event EventTick (DeltaSeconds)
  (bind pc (Game|GetPlayerController 0))
  (bind springarm (Variables|Camera|GetSpringArm))
  (bind armLen (Class|SpringArmComponent|GetTargetArmLength springarm))
  (bind zoomSpeed (Variables|Default|GetZoomSpeed))
  (if (Variables|Default|GetIsDropping)
    (Utilities|IsValid pc
      (:"Is Valid"
        (bind (hit didhit) (Game|Player|GetHitResultUnderCursorByChannel pc "Visibility" true))
        (if didhit
          (bind (bBlockingHit bInitialOverlap Time Distance Location ImpactPoint Normal ImpactNormal PhysMat HitActor) (Collision|BreakHitResult hit))
          (Pawn|Input|AddMovementInput self (Math|Vector|GetUnitDirection(Vector) (Math|Vector|MakeVector (.x (Transformation|GetActorLocation)) (.y (Transformation|GetActorLocation))) (Math|Vector|MakeVector (.x ImpactPoint) (.y ImpactPoint))) (Variables|Default|GetSteerScale)))
        (if (Game|Player|IsInputKeyDown pc "SpaceBar")
          (Class|MovementComponent|SetVelocity (Math|Vector|MakeVector (.x (Transformation|GetVelocity)) (.y (Transformation|GetVelocity)) (Variables|Default|GetGlideFallSpeed)) (Variables|Character|GetCharacterMovement)))))
    (else
      (Variables|Default|SetAbilityDelta DeltaSeconds)
      (CallFunction|TickAbilities)
      (Utilities|IsValid pc
        (:"Is Valid"
          (if (Game|Player|WasInputKeyJustPressed pc "MouseScrollUp")
            (Class|SpringArmComponent|SetTargetArmLength (select (< (- armLen zoomSpeed) (Variables|Default|GetMinZoom)) (Variables|Default|GetMinZoom) (- armLen zoomSpeed)) springarm))
          (if (Game|Player|WasInputKeyJustPressed pc "MouseScrollDown")
            (Class|SpringArmComponent|SetTargetArmLength (select (> (+ armLen zoomSpeed) (Variables|Default|GetMaxZoom)) (Variables|Default|GetMaxZoom) (+ armLen zoomSpeed)) springarm))
          (if (> (Variables|Default|GetPendingAbility) 0)
            (CallFunction|UpdateAbilityAim)
            (if (or (Game|Player|WasInputKeyJustPressed pc "RightMouseButton") (Game|Player|WasInputKeyJustPressed pc "Escape"))
              (CallFunction|CancelAbilityAim))
            (else
              (if (and (Game|Player|WasInputKeyJustPressed pc "Q") (<= (Variables|Default|GetCDQ) 0.0))
                (CallFunction|BeginAbilityAim 1))
              (if (and (Game|Player|WasInputKeyJustPressed pc "W") (<= (Variables|Default|GetCDW) 0.0))
                (CallFunction|BeginAbilityAim 2))
              (if (and (Game|Player|WasInputKeyJustPressed pc "E") (<= (Variables|Default|GetCDE) 0.0))
                (CallFunction|BeginAbilityAim 3))
              (if (and (Game|Player|WasInputKeyJustPressed pc "R") (<= (Variables|Default|GetCDR) 0.0))
                (CallFunction|BeginAbilityAim 4)))))))))

(event EventBeginPlay
  (bind pc (Game|GetPlayerController 0))
  (bind movement (Variables|Character|GetCharacterMovement))
  (bind springarm (Variables|Camera|GetSpringArm))
  (Variables|Default|SetIsDropping true)
  (Transformation|SetActorLocation self (Math|Vector|MakeVector (.x (Transformation|GetActorLocation)) (.y (Transformation|GetActorLocation)) (Variables|Default|GetDropHeight)) false true)
  (Class|CharacterMovementComponent|SetGravityScale (Variables|Default|GetFreefallGravityScale) movement)
  (Class|CharacterMovementComponent|SetAirControl (Variables|Default|GetAirControlStrength) movement)
  (Variables|Default|SetSavedArmLength (Class|SpringArmComponent|GetTargetArmLength springarm))
  (Class|SpringArmComponent|SetTargetArmLength (Variables|Default|GetSavedArmLength) springarm)
  (Transformation|SetRelativeRotation springarm (Math|Rotator|MakeRotator 0.0 -90.0 0.0))
  (Utilities|IsValid pc
    (:"Is Valid"
      (Class|PlayerController|SetShowMouseCursor true pc)
      (Class|PlayerController|SetEnableClickEvents true pc)
      (Class|PlayerController|SetEnableMouseOverEvents true pc)
      (bind ctrl (Utilities|Casting|CastToBP_TopDownController (Pawn|GetController))
        (:then
          (Class|BPTopDownController|SetDropMode true ctrl)
          (Development|PrintString "SKY DROP - aim with mouse, hold SPACE to plane"))
        (:CastFailed
          (Development|PrintString "SKY DROP - aim with mouse, hold SPACE to plane"))))))

(event EventOnLanded (Hit)
  (bind movement (Variables|Character|GetCharacterMovement))
  (bind springarm (Variables|Camera|GetSpringArm))
  (if (Variables|Default|GetIsDropping)
    (Variables|Default|SetIsDropping false)
    (Class|CharacterMovementComponent|SetGravityScale (Variables|Default|GetDefaultGravityScale) movement)
    (Class|CharacterMovementComponent|SetAirControl (Variables|Default|GetDefaultAirControl) movement)
    (Class|SpringArmComponent|SetTargetArmLength (Variables|Default|GetSavedArmLength) springarm)
    (Transformation|SetRelativeRotation springarm (Math|Rotator|MakeRotator 0.0 (Variables|Default|GetTopDownArmPitch)))
    (bind ctrl (Utilities|Casting|CastToBP_TopDownController (Pawn|GetController))
      (:then
        (Class|BPTopDownController|SetDropMode false ctrl)
        (Development|PrintString "LANDED - QWER to aim, LMB to cast, RMB to cancel"))
      (:CastFailed
        (Development|PrintString "LANDED - QWER to aim, LMB to cast, RMB to cancel")))))
