(fn SpawnEnemy ()
  (if (Variables|Default|GetIsSpawningWave)
    (if (< (Variables|Default|GetWaveSpawnedCount) (Variables|Default|GetEnemiesPerWave))
      (CallFunction|SpawnEnemyInner)
      (if (>= (Variables|Default|GetWaveSpawnedCount) (Variables|Default|GetEnemiesPerWave))
        (Utilities|Time|ClearTimerbyFunctionName self "SpawnEnemy")
        (Variables|Default|SetIsSpawningWave false)
        (Variables|Default|SetWaitingForClear true)
        (Development|PrintString "Wave spawning done - defeat remaining enemies!" true true "(R=1.000000,G=0.700000,B=0.200000,A=1.000000)" 3.0)
        (Utilities|Time|SetTimerbyFunctionName self "CheckWaveClear" 0.5 true)))))
