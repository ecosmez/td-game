(fn AdvanceAlongPath (DeltaSeconds)
  (bind _self self)
  (bind _returnvalue (Transformation|GetActorLocation _self))
  (bind _slowfactor (Variables|Default|GetSlowFactor))
  (bind _movespeed (Variables|Default|GetMoveSpeed))
  (bind _returnvalue_1 (* _movespeed _slowfactor))
  (bind _lastdelta (Variables|Default|GetLastDelta))
  (bind _waypointindex (Variables|Default|GetWaypointIndex))
  (bind _waypoints (Variables|Default|GetWaypoints))
  (bind _output (Utilities|Array|Get(acopy) _waypoints _waypointindex))
  (bind _returnvalue_2 (Math|Interpolation|VInterptoConstant _returnvalue _output _lastdelta _returnvalue_1))
  (bind _crystalactor (Variables|Default|GetCrystalActor))
  (bind _returnvalue_3 (Transformation|GetActorLocation _crystalactor))
  (bind _returnvalue_4 (Math|Interpolation|VInterptoConstant _returnvalue _returnvalue_3 _lastdelta _returnvalue_1))
  (bind _attackcooldown (Variables|Default|GetAttackCooldown))
  (bind _returnvalue_5 (- _attackcooldown DeltaSeconds))
  (bind _flat_waypoint (Math|Vector|MakeVector :X (.x _output) :Y (.y _output) :Z (.z _returnvalue)))
  (bind _flat_crystal (Math|Vector|MakeVector :X (.x _returnvalue_3) :Y (.y _returnvalue_3) :Z (.z _returnvalue)))
  (if (Variables|Default|GetIsAttackingCrystal)
    (if (<= (Variables|Default|GetStunRemaining) 0.0)
      (Variables|Default|SetAttackCooldown _returnvalue_5)
      (if (<= _returnvalue_5 0.0)
        (CallFunction|AttackCrystal))))
  (if (not (Variables|Default|GetIsAttackingCrystal))
    (if (Variables|Default|GetIsWallBlocked)
      (if (<= (Variables|Default|GetStunRemaining) 0.0)
        (CallFunction|AttackBlockingWall :DeltaSeconds DeltaSeconds))
      (else
        (if (<= (Variables|Default|GetStunRemaining) 0.0)
          (if (Utilities|Array|IsEmpty (Variables|Default|GetWaypoints))
            (CallFunction|ChoosePath))
          (if (Utilities|Array|IsValidIndex _waypoints _waypointindex)
            (Transformation|SetActorLocation _self _returnvalue_2 false true)
            (Transformation|SetActorRotation _self (Math|Rotator|FindLookatRotation _returnvalue _flat_waypoint) true)
            (if (<= (Math|Vector|Distance(Vector) _returnvalue_2 _output) 120.0)
              (CallFunction|AdvanceWaypoint))
            (else
              (Utilities|IsValid _crystalactor
                (:"Is Valid"
                  (Transformation|SetActorLocation _self _returnvalue_4 false true)
                  (Transformation|SetActorRotation _self (Math|Rotator|FindLookatRotation _returnvalue _flat_crystal) true)
                  (if (<= (Math|Vector|Distance(Vector) _returnvalue_4 _returnvalue_3) 200.0)
                    (CallFunction|ReachCrystal)))
                (:"Is Not Valid"
                  (CallFunction|ReachCrystal))))))))))
