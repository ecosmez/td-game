(fn CancelTowerSelection ()
  (if (or (Variables|Default|GetTowerDragLive) (Variables|Default|GetHasTowerSelected))
    (Variables|Default|SetIsDragging false)
    (Variables|Default|SetTowerDragLive false)
    (Variables|Default|SetHasTowerSelected false)
    (CallFunction|DestroyGhostTower)
    (CallFunction|HideHexGrid)
    (if (Variables|Default|GetWavesStarted)
      (Variables|Default|SetStatusLine "WAVES STARTED - keep building!")
      (Variables|Default|SetHintLine "Drag towers anytime | ESC cancels | Defend the crystal")
      (else
        (Variables|Default|SetStatusLine "BUILD PHASE - drag tower from the UI")
        (Variables|Default|SetHintLine "Drag BASIC TOWER onto a hex | ENTER starts waves | ESC cancels")))
    (Development|PrintString "Tower build canceled" true true "(R=1.000000,G=0.700000,B=0.200000,A=1.000000)" 1.5)))
