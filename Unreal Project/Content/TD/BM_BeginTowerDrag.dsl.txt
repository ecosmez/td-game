(fn BeginTowerDrag ()
  (if (Variables|Default|GetIsBuildPhase)
    (Variables|Default|SetIsDragging true)
    (Variables|Default|SetTowerDragLive true)
    (Variables|Default|SetHasTowerSelected true)
    (Variables|Default|SetStatusLine "Dragging BASIC TOWER - release on a hex")
    (Variables|Default|SetHintLine "Drop on a glowing hex | ESC / RMB cancels")
    (CallFunction|ShowHexGrid)
    (CallFunction|SpawnGhostTower)
    (Development|PrintString "Dragging BASIC TOWER" true true "(R=0.200000,G=0.900000,B=1.000000,A=1.000000)" 1.5)))
