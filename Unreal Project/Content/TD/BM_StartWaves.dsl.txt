(fn StartWaves ()
  (if (Variables|Default|GetIsBuildPhase)
    (Variables|Default|SetIsBuildPhase false)
    (Variables|Default|SetHasTowerSelected false)
    (Variables|Default|SetIsDragging false)
    (Variables|Default|SetStatusLine "WAVES STARTED")
    (Variables|Default|SetHintLine "Defend the crystal!")
    (CallFunction|HideBuildHUD)
    (Development|PrintString "Build phase over - waves starting!" true true "(R=1.000000,G=0.850000,B=0.100000,A=1.000000)" 3.0)
    (Actor|GetAllActorsOfClass "/Game/TD/BP_EnemySpawner.BP_EnemySpawner_C"
      (:then
        (if (Utilities|Array|IsEmpty _outactors)
          (Development|PrintString "No spawner in level" true true "(R=1.000000,G=0.000000,B=0.000000,A=1.000000)" 5.0)
          (else
            (bind spawner (Utilities|Array|RandomArrayItem _outactors))
            (Utilities|IsValid spawner
              (:"Is Valid"
                (bind sp (Utilities|Casting|CastToBP_EnemySpawner spawner)
                  (:then
                    (Class|BPEnemySpawner|StartWaves sp))
                  (:CastFailed
                    (Development|PrintString "Spawner cast failed"))))
              (:"Is Not Valid"
                (Development|PrintString "Spawner invalid")))))))))
